\section{Proof of \texorpdfstring{\cref{lem:sum_norm_g_over_mu_geq_2sqrt_leq_log}}{Lemma \ref{lem:sum_norm_g_over_mu_geq_2sqrt_leq_log}}}\label{sec:proof_of_sum_norm_g_over_mu_geq_2sqrt_leq_log}

The proof of \cref{lem:sum_norm_g_over_mu_geq_2sqrt_leq_log} is similar to existing results~\citep[Lemma~3.2]{wardAdaGradStepsizesSharp2020},~\citep[Lemma~3.1]{Gratton08022024}.
\begin{proof}
    Recall that \cref{eq:offo_based_update} defines $\mu_k$ as $\theta_k \sqrt{\varsigma + \sum_{j \in K^+, \, j \leq k} \norm{g_j}^2}$, and \refLine{line:update_theta} of \cref{alg:regularized_qn} asserts $\theta_k \in [\theta_{\min}, \theta_{\max}]$.
    Let $k_i \in K^+$ be the $i$-th element of $K^+$.
    For any $0 < x < y$, $x/y \le -\ln(1-x/y) = \ln y - \ln(y-x)$ holds. Thus, we have
    \begin{align*}
        \frac{\norm{g_{k_i}}^2}{\mu_{k_i}^2}
         & \leq \frac{1}{\theta_{\min}^2} \frac{\norm{g_{k_i}}^2}{\varsigma+\sum_{\ell=0}^{i} \norm{g_{k_\ell}}^2}                                                    &  & (\text{definition})        \\
         & \le \frac{1}{\theta_{\min}^2} \qty(\ln\qty(\varsigma+\sum_{\ell=0}^{i} \norm{g_{k_\ell}}^2) - \ln\qty(\varsigma+\sum_{\ell=0}^{i-1} \norm{g_{k_\ell}}^2)). &  & (x/y \le \ln y - \ln(y-x))
    \end{align*}
    Similarly, $x/\sqrt{y} \ge x/(\sqrt{y}+\sqrt{y-x}) = \sqrt{y}-\sqrt{y-x}$ holds. Thus, we have
    \begin{align*}
        \frac{\norm{g_{k_i}}^2}{\mu_{k_i}}
         & \ge \MPCchanged{\frac{1}{\theta_{\max}} \frac{\norm{g_{k_i}}^2}{\sqrt{\varsigma + \sum_{\ell=0}^{i} \norm{g_{k_\ell}}^2}}}                                                      &  & (\text{definition})                  \\
         & \ge \frac{1}{\theta_{\max}} \qty(\sqrt{\varsigma+\sum_{\ell=0}^{i\vphantom{i-1}} \norm{g_{k_\ell}}^2} - \sqrt{\varsigma+\sum_{\ell=0}^{i-1} \norm{g_{k_\ell}}^2}). &  & (x/\sqrt{y} \ge \sqrt{y}-\sqrt{y-x})
    \end{align*}
    Summing these inequalities over $0 \le i < \abs{K^+}$ yields the claim.
    \myQED
\end{proof}

\section{Analysis of the Accumulator-Restart Heuristic}\label{sec:analysis_accumulator_restart}

We analyze the optional variant described in \cref{eq:restart_condition}, which resets only the accumulator in \cref{eq:offo_based_update} when the decrease relative to the reference value reaches a fixed threshold $\tau>0$.
The sets $K^0$ and $K^+$ and the reference values used in \cref{eq:restart_condition} are retained after a restart.
For $k \geq 1$, define
\begin{equation}\label{eq:restart_reference_value}
    q_k \defeq \min_{j \in K^0,\,j<k} \qty(\overline{f}(x_j)-\Delta_j).
\end{equation}
Thus, a restart at iteration $k$ occurs only if $q_k-\overline{f}(x_k)\geq\tau$.

\begin{proposition}\label{prop:finite_accumulator_restarts}
    Under \cref{ass:bounded,ass:LSmooth,ass:BBounds}, the number $R$ of accumulator restarts satisfying \cref{eq:restart_condition} is finite and satisfies
    \begin{equation}\label{eq:restart_count_bound}
        R \leq \frac{\overline{f}(x_0)-\overline{f}^*+\Delta_{\max}}{\tau}.
    \end{equation}
\end{proposition}
\begin{proof}
    Since $k=0$ belongs to $K^0$, \cref{eq:restart_reference_value} gives $q_1=\overline{f}(x_0)-\Delta_0\leq\overline{f}(x_0)$.
    By \cref{eq:ass_bounded_overline_f} and the bound $\Delta_k\leq\Delta_{\max}$ for $k\in K^0$ established in the proof of \cref{lem:delta_bound_and_sum}, we have
    \begin{equation}\label{eq:restart_reference_lower_bound}
        q_k \geq \overline{f}^*-\Delta_{\max}.
    \end{equation}
    Let $r_1<r_2<\cdots<r_R$ be the restart indices.
    Since \cref{eq:restart_condition} implies $r_i\in K^0$, updating the reference value after iteration $r_i$ gives
    \begin{align}
        q_{r_i+1}
         & \leq \overline{f}(x_{r_i})-\Delta_{r_i}             &  & (\text{\cref{eq:restart_reference_value}}) \notag \\
         & \leq q_{r_i}-\tau-\Delta_{r_i}                      &  & (\text{\cref{eq:restart_condition}})       \notag \\
         & \leq q_{r_i}-\tau.                                  &  & (\Delta_{r_i}\geq0)                         \label{eq:restart_reference_decrease}
    \end{align}
    The sequence $(q_k)$ is non-increasing by \cref{eq:restart_reference_value}.
    Applying \cref{eq:restart_reference_decrease} recursively at the restart indices and using $q_1\leq\overline{f}(x_0)$ yields
    \begin{equation}\label{eq:restart_reference_recursive_bound}
        q_{r_R+1}\leq q_1-R\tau\leq\overline{f}(x_0)-R\tau.
    \end{equation}
    Combining \cref{eq:restart_reference_lower_bound,eq:restart_reference_recursive_bound} and rearranging gives \cref{eq:restart_count_bound}.
    This completes the proof.
    \myQED
\end{proof}

We next show that these finitely many restarts preserve the convergence rate in \cref{thm:gc}.
Let $r_1<r_2<\cdots<r_R$ again denote the restart indices, and set $r_0=-1$ and $r_{R+1}=k_{\max}$.
For $i=0,1,\ldots,R$, define the $i$-th phase and its sum of squared gradient norms by
\begin{equation}\label{eq:restart_phase_definition}
    K_i^+ \defeq \{k\in K^+\mid r_i<k<r_{i+1}\}, \qquad S_i \defeq \sum_{k\in K_i^+}\norm{g_k}^2.
\end{equation}
Within each $K_i^+$, the accumulator is not reset and therefore satisfies
\begin{equation}\label{eq:offo_based_update_in_restart_phase}
    \mu_k=\theta_k\sqrt{\varsigma+\sum_{j\in K_i^+,\,j\leq k}\norm{g_j}^2}, \qquad \theta_k\in[\theta_{\min},\theta_{\max}].
\end{equation}

\begin{theorem}\label{thm:gc_with_accumulator_restarts}
    Under \cref{ass:exact_gradient,ass:bounded,ass:LSmooth,ass:BBounds}, suppose that only the accumulator in \cref{eq:offo_based_update} is reset according to \cref{eq:restart_condition} with a fixed threshold $\tau>0$.
    Then, the average of squared gradient norms satisfies
    \begin{equation*}
        \frac{1}{k_{\max}}\sum_{k=0}^{k_{\max}-1}\norm{g_k}^2=\order{\frac{1}{k_{\max}}}.
    \end{equation*}
\end{theorem}
\begin{proof}
    Applying the proof of \cref{lem:sum_norm_g_over_mu_geq_2sqrt_leq_log} to \cref{eq:offo_based_update_in_restart_phase} and then following the proof of \cref{lem:sum_small_delta_k_bound_Kplus} for each $K_i^+$ yield
    \begin{equation}\label{eq:restart_phase_delta_bound}
        \sum_{k\in K_i^+}\delta_k
        \geq C_1\qty(\sqrt{\varsigma+S_i}-\sqrt{\varsigma})
        -C_2\qty(\ln\qty(\varsigma+S_i)-\ln(\varsigma)).
    \end{equation}
    Restarting the accumulator does not alter $K^0$ or \cref{eq:mu_0_condition}, so \cref{lem:sum_small_delta_k_bound_K0} remains valid.
    Combining \cref{lem:sum_small_delta_k_bound_K0,eq:restart_phase_delta_bound} with $f(x_0)-f(x_{k_{\max}})=\sum_{k\in K}\delta_k$ and $f(x_{k_{\max}})\geq f^*$ gives
    \begin{equation}\label{eq:restart_combined_bound}
        F_R
        \geq \frac{c\overline{\alpha}}{M}\sum_{k\in K^0}\norm{g_k}^2
        +\sum_{i=0}^{R}\qty(C_1\sqrt{\varsigma+S_i}-C_2\ln\qty(\varsigma+S_i)),
    \end{equation}
    where
    \begin{equation}\label{eq:def_F_R}
        F_R\defeq f(x_0)-f^*+\frac{2}{1-\feps}\qty(\overline{f}(x_0)-\overline{f}^*+\Delta_{\max})+(R+1)\qty(C_1\sqrt{\varsigma}-C_2\ln(\varsigma)).
    \end{equation}
    Recall the function $\phi$ and its minimizer $G^*$ defined in the proof of \cref{thm:gc}.
    Since $\sqrt{\varsigma + S_i} \geq \sqrt{S_i}$ and $\phi(G) \geq \phi(G^*)$ for all $G > 0$, the following holds for every $i=0,1,\ldots,R$:
    \begin{align}
        C_1\sqrt{\varsigma+S_i}-C_2\ln\qty(\varsigma+S_i)
         & =\frac{C_1}{2}\sqrt{\varsigma+S_i}+\phi\qty(\varsigma+S_i)       &  & (\text{\cref{eq:phi_G}}) \notag \\
         & \geq\frac{C_1}{2}\sqrt{S_i}+\phi(G^*).                            &  & \label{eq:restart_phase_phi_bound}
    \end{align}
    Substituting \cref{eq:restart_phase_phi_bound} into \cref{eq:restart_combined_bound} and using $\sum_{i=0}^R\sqrt{S_i}\geq\sqrt{\sum_{i=0}^R S_i}$ yield
    \begin{equation}\label{eq:restart_final_inequality}
        0\leq\frac{c\overline{\alpha}}{M}\sum_{k\in K^0}\norm{g_k}^2
        +\frac{C_1}{2}\sqrt{\sum_{k\in K^+}\norm{g_k}^2}
        \leq F_R',
    \end{equation}
    where $F_R'\defeq F_R-(R+1)\phi(G^*)$.
    Applying the same maximization argument as in the proof of \cref{thm:gc} to \cref{eq:restart_final_inequality} gives
    \begin{equation}\label{eq:gc_result_with_accumulator_restarts}
        \frac{1}{k_{\max}}\sum_{k=0}^{k_{\max}-1}\norm{g_k}^2
        \leq\frac{1}{k_{\max}}\max\qty(\frac{MF_R'}{c\overline{\alpha}},\frac{4(F_R')^2}{C_1^2}).
    \end{equation}
    By \cref{prop:finite_accumulator_restarts}, $R$ is bounded above by a constant independent of $k_{\max}$ for every fixed $\tau>0$.
    Hence, $F_R'$ is also independent of $k_{\max}$, and \cref{eq:gc_result_with_accumulator_restarts} yields the stated rate.
    This completes the proof.
    \myQED
\end{proof}
\end{MPCchange}

\section{Proof of \texorpdfstring{\cref{prop:bounded_BFGS}}{Proposition \ref{prop:bounded_BFGS}}}\label{sec:proof_of_bounded_BFGS}

The proof of \cref{prop:bounded_BFGS} is similar to existing results~\citep[Theorem 5.5]{gaoQuasiNewtonMethodsSuperlinear2018}~\citep[Lemma 1]{gowerStochasticBlockBFGS2016}.
\begin{proof}
    We consider standard BFGS updates with a single curvature pair $(s,y)$.
    From \cref{eq:curvature_conditions1,eq:curvature_conditions2}, we have
    \begin{equation}\label{eq:curvature_norm_bounds}
        \norm{\frac{y y^\top}{y^\top s}} = \frac{y^\top y}{y^\top s} \le \Lambda, \quad
        \norm{\frac{y s^\top}{y^\top s}} \le \frac{\norm{y}\norm{s}}{\sqrt{\frac{1}{\Lambda}\norm{y}^2}\sqrt{\lambda\norm{s}^2}} = \sqrt{\kappa}, \quad
        \norm{\frac{s s^\top}{y^\top s}} = \frac{s^\top s}{y^\top s} \le \frac{1}{\lambda}.
    \end{equation}
    We next bound the scalar $\gamma$ used in the initial matrix $\gamma I$.
    From \cref{eq:curvature_conditions1}, we have $\gamma \le \Lambda$.
    By Cauchy--Schwarz inequality and \cref{eq:curvature_conditions2}, we have $\gamma \ge \norm{y_{k_0}}^2 / \norm{y_{k_0}}\norm{s_{k_0}} = \norm{y_{k_0}} / \norm{s_{k_0}} \ge \lambda$. Thus, $\gamma$ is bounded by $\lambda \le \gamma \le \Lambda$.

    Let $B, H$ denote the positive definite Hessian approximation and its inverse before the update, and $B_+, H_+$ the updated matrices.
    The BFGS update rule is
    \begin{equation*}
        B_+   =B - \frac{Bss^\top B}{s^\top Bs} + \frac{yy^\top}{y^\top s}, \quad
        H_+   =\left(I-\frac{s y^\top}{y^\top s}\right)H\left(I-\frac{y s^\top}{y^\top s}\right)+\frac{s s^\top}{y^\top s}.
    \end{equation*}
    Since $B$ is a symmetric positive definite matrix and $s \neq 0$, we have
    \begin{align}
        \norm{B_+} & \le\norm{B}+\norm{\frac{yy^\top}{y^\top s}} \le \norm{B}+\Lambda,\label{eq:BFGS_B_norm_bound}                                                                                                                                                     \\
        \norm{H_+} & =\norm{\qty(I-\frac{y s^\top}{y^\top s})^\top H\qty(I-\frac{y s^\top}{y^\top s})+\frac{s s^\top}{y^\top s}}                                                                                \notag                                                 \\
                   & \le\norm{I-\frac{y s^\top}{y^\top s}}^2\norm{H}+\norm{\frac{s s^\top}{y^\top s}}                                                                                                       \notag                                                     \\
                   & \le\qty(1+\norm{\frac{y s^\top}{y^\top s}})^2\norm{H}+\norm{\frac{s s^\top}{y^\top s}}                                                                                                       \notag                                               \\
                   & \le \qty(1+\sqrt{\kappa})^2\norm{H}+\frac{1}{\lambda}.  \label{eq:BFGS_H_norm_bound}                                                                                                                &  & (\text{\cref{eq:curvature_norm_bounds}})
    \end{align}
    Recall that $\mathrm{BFGS}(\{(s_{k_i},y_{k_i})\}_{i=0}^{p-1})$ denotes the matrix obtained by sequentially applying the BFGS update to the initial matrix $\gamma I$ using the curvature pairs $\{(s_{k_i},y_{k_i})\}_{i=0}^{p-1}$.
    Applying \cref{eq:BFGS_B_norm_bound,eq:BFGS_H_norm_bound} recursively over $p$ updates and using the initial matrix $\gamma I$, we obtain
    \begin{align*}
        \norm{\mathrm{BFGS}(\{s_{k_i},y_{k_i}\}_{i=0}^{p-1})}        & \le \gamma + p\Lambda \le (1+p) \Lambda,                                                                               \\
        \norm{(\mathrm{BFGS}(\{s_{k_i},y_{k_i}\}_{i=0}^{p-1}))^{-1}} & \le (1+\sqrt{\kappa})^{2p} \gamma^{-1} +\frac{1}{\lambda}\sum_{i=0}^{p-1}(1+\sqrt{\kappa})^{2i}                        \\
                                                                     & \le (1+\sqrt{\kappa})^{2p} \frac{1}{\lambda}+\frac{1}{\lambda}\frac{(1+\sqrt{\kappa})^{2p}-1}{(1+\sqrt{\kappa})^{2}-1} \\
                                                                     & \le (1+\sqrt{\kappa})^{2p}\qty(\frac{1}{\lambda}+\frac{1}{\lambda\qty(2\sqrt{\kappa}+\kappa)}).
    \end{align*}
    Therefore, $\mathrm{BFGS}(\{s_{k_i},y_{k_i}\}_{i=0}^{p-1})$ satisfies the stated bounds in \cref{eq:bounded_BFGS_mM}.
    \myQED
\end{proof}

The two curvature inequalities are standard sufficient conditions for bounded limited-memory BFGS matrices~\citep[Theorem 2.1]{byrdToolAnalysisQuasiNewton1989}.
